+# ============================================================================
+# DATA MODELS

+
+@dataclass
+class FileChange:
+    """Represents changes in a single file"""
+    filename: str
+    old_path: Optional[str]
+    new_path: str
+    change_type: str
+    added_lines: int
+    deleted_lines: int
+    diff: str
+    complexity: int
+    methods_changed: List[str]

+@dataclass
+class CommitAnalysis:
+    """Stores complete analysis of commit differences"""
+    repo_path: str
+    commit_from: str
+    commit_to: str
+    commit_from_hash: str
+    commit_to_hash: str
+    commit_from_message: str
+    commit_to_message: str
+    commit_from_author: str
+    commit_to_author: str
+    commit_from_date: datetime
+    commit_to_date: datetime
+    total_files_changed: int
+    total_lines_added: int
+    total_lines_deleted: int

+# ============================================================================
+# REPOSITORY VALIDATOR
+# ============================================================================
+
+class RepositoryValidator:
+    """Validates repository and commit inputs"""
+    
+    @staticmethod
+    def validate_repo_path(repo_path: str) -> bool:
+        """Check if the repository path is valid"""
+        if not os.path.exists(repo_path):
+            raise ValueError(f"Repository path does not exist: {repo_path}")
+        
+        git_dir = os.path.join(repo_path, '.git')
+        if not os.path.exists(git_dir):
+            raise ValueError(f"Not a valid git repository: {repo_path}")
+        
+        return True
+    
+    @staticmethod
+    def get_commit_object(repo_path: str, commit_id: str) -> Commit:
+        """Retrieve commit object from repository"""
+        try:
+            for commit in Repository(repo_path, single=commit_id).traverse_commits():
+                return commit
+        except Exception as e:
+            raise ValueError(f"Invalid commit identifier '{commit_id}': {str(e)}")
+
+
+# ============================================================================
+# COMMIT DIFFERENCE ANALYZER
+# ============================================================================
+
+class CommitDifferenceAnalyzer:
+    """Analyzes differences between two commits"""
+    
+    def __init__(self, repo_path: str):
+        self.repo_path = repo_path
+        self.validator = RepositoryValidator()
+        self.validator.validate_repo_path(repo_path)
+    
+    def get_commit_info(self, commit_id: str) -> Tuple[str, str, str, datetime]:
+        """Extract commit information"""
+        commit = self.validator.get_commit_object(self.repo_path, commit_id)
+        return (
+            commit.hash,
+            commit.msg,
+            commit.author.name,
+            commit.author_date
+        )
+    
+    def analyze_commits(self, commit_from: str, commit_to: str) -> CommitAnalysis:
+        """Perform comprehensive analysis between two commits"""
+        
+        from_hash, from_msg, from_author, from_date = self.get_commit_info(commit_from)
+        to_hash, to_msg, to_author, to_date = self.get_commit_info(commit_to)
+        
+        file_changes = []
+        total_added = 0
+        total_deleted = 0
+        
+        commits = list(Repository(
+            self.repo_path,
+            from_commit=commit_from,
+            to_commit=commit_to
+        ).traverse_commits())
+        
+        files_dict = {}
+        for commit in commits:
+            for modified_file in commit.modified_files:
+                filename = modified_file.new_path or modified_file.old_path
+                
+                if filename not in files_dict:
+                    files_dict[filename] = {
+                        'added': 0,
+                        'deleted': 0,
+                        'diff': [],
+                        'change_type': modified_file.change_type.name,
+                        'old_path': modified_file.old_path,
+                        'new_path': modified_file.new_path,
+                        'complexity': modified_file.complexity if hasattr(modified_file, 'complexity') else 0,
+                        'methods': set()
+                    }
+                
+                files_dict[filename]['added'] += modified_file.added_lines
+                files_dict[filename]['deleted'] += modified_file.deleted_lines
+                
+                if modified_file.diff:
+                    files_dict[filename]['diff'].append(modified_file.diff)
+                
+                if hasattr(modified_file, 'changed_methods'):
+                    for method in modified_file.changed_methods:
+                        files_dict[filename]['methods'].add(method.name)
+        
+        for filename, data in files_dict.items():
+            file_change = FileChange(
+                filename=filename,
+                old_path=data['old_path'],
+                new_path=data['new_path'],
+                change_type=data['change_type'],
+                added_lines=data['added'],
+                deleted_lines=data['deleted'],
+                diff='\n'.join(data['diff']),
+                complexity=data['complexity'],
+                methods_changed=list(data['methods'])
+            )
+            file_changes.append(file_change)
+            total_added += data['added']
+            total_deleted += data['deleted']
+        
+        return CommitAnalysis(
+            repo_path=self.repo_path,
+            commit_from=commit_from,
+            commit_to=commit_to,
+            commit_from_hash=from_hash,
+            commit_to_hash=to_hash,
+            commit_from_message=from_msg,
+            commit_to_message=to_msg,
+            commit_from_author=from_author,
+            commit_to_author=to_author,
+            commit_from_date=from_date,
+            commit_to_date=to_date,
+            total_files_changed=len(file_changes),
+            total_lines_added=total_added,
+            total_lines_deleted=total_deleted,
+            file_changes=file_changes
+        )
+

+        self.prompt = None

+        """Generate prompt for Gemini Code Assist"""

+        self.prompt = generator.generate_prompt()

+    
+    def save_results(self, output_dir: str = "./output"):
+        """Save all results to files"""

+            raise RuntimeError("Must run analysis and generate prompt first")
+        
+        os.makedirs(output_dir, exist_ok=True)
+        
+        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
+        
+        prompt_file = os.path.join(
+            output_dir,
+            f"gemini_prompt_{timestamp}.txt"
+        )

+        """Display summary to console"""
+        if not self.analysis:
